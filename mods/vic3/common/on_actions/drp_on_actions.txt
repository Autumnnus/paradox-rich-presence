# Paradox Rich Presence
# Writes the player countries' status to debug.log; the companion app reads these lines
# and forwards them to Discord. Has no effect on the game itself.
#
# Protocol (version 2):
#   DRP|2|local|<local player's tag>
#   DRP|2|begin|<tag>
#   DRP|2|kv|<tag>|<key>|<value>      single-value field
#   DRP|2|item|<tag>|<key>|<value>    list item (same key may repeat)
#   DRP|2|end|<tag>
# Data functions that emit tooltip markup (such as GetRank) are avoided; fixed keys are
# written through triggers instead and translated by the companion.

on_game_started_after_lobby = {
	on_actions = { drp_report_players }
}

on_monthly_pulse = {
	on_actions = { drp_report_players }
}

drp_report_players = {
	effect = {
		debug_log = "DRP|2|local|[GetPlayer.GetTagName]"

		every_country = {
			limit = { is_player = yes }
			save_scope_as = drp_player

			debug_log = "DRP|2|begin|[THIS.GetCountry.GetTagName]"

			# General
			debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|country|[THIS.GetCountry.GetNameNoFormatting]"
			debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|date|[TimeKeeper.GetCurrentDate.GetString]"

			# Economy
			debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|gdp|[THIS.GetCountry.GetGDP|1*]"
			debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|gdp_rank|[THIS.GetCountry.GetGDPRanking]"
			debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|population|[THIS.GetCountry.GetTotalPopulation|*]"

			# Election (status icon)
			if = {
				limit = { in_election_campaign = yes }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|election|yes"
			}

			# Rank: written as a key because GetRank returns tooltip markup
			if = {
				limit = { country_rank = rank_value:great_power }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|rank|great_power"
			}
			else_if = {
				limit = { country_rank = rank_value:major_power }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|rank|major_power"
			}
			else_if = {
				limit = { country_rank = rank_value:minor_power }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|rank|minor_power"
			}
			else_if = {
				limit = { country_rank = rank_value:insignificant_power }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|rank|insignificant_power"
			}
			else_if = {
				limit = { country_rank = rank_value:unrecognized_major_power }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|rank|unrecognized_major_power"
			}
			else_if = {
				limit = { country_rank = rank_value:unrecognized_regional_power }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|rank|unrecognized_regional_power"
			}
			else_if = {
				limit = { country_rank = rank_value:unrecognized_power }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|rank|unrecognized_power"
			}
			else_if = {
				limit = { country_rank = rank_value:decentralized_power }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|rank|decentralized_power"
			}

			# War and diplomacy
			if = {
				limit = { is_at_war = yes }
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|war|yes"
			}
			else = {
				debug_log = "DRP|2|kv|[THIS.GetCountry.GetTagName]|war|no"
			}
			every_country = {
				limit = { has_war_with = scope:drp_player }
				debug_log = "DRP|2|item|[SCOPE.sCountry('drp_player').GetTagName]|enemy|[THIS.GetCountry.GetNameNoFormatting]"
				if = {
					limit = { is_revolutionary = yes }
					debug_log = "DRP|2|item|[SCOPE.sCountry('drp_player').GetTagName]|enemy_rev|[THIS.GetCountry.GetNameNoFormatting]"
				}
			}
			if = {
				limit = { is_diplomatic_play_committed_participant = yes }
				every_country = {
					limit = {
						is_diplomatic_play_enemy_of = scope:drp_player
						NOT = { has_war_with = scope:drp_player }
					}
					debug_log = "DRP|2|item|[SCOPE.sCountry('drp_player').GetTagName]|play_enemy|[THIS.GetCountry.GetNameNoFormatting]"
				}
			}

			debug_log = "DRP|2|end|[THIS.GetCountry.GetTagName]"
		}
	}
}
